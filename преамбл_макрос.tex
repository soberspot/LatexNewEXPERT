% Фото (автоматическая нумерация и шаблон-заглушка)
\newcommand{\insertphoto}[2]{%
	\begin{figure}[H]
		\centering
		\IfFileExists{images/#1}{
			\includegraphics[width=0.85\textwidth]{images/#1}
		}{
			\begin{tikzpicture}
				\draw[gray, thick] (0,0) rectangle (12,8);
				\node[gray!70!black, align=center] at (6,4)
				{\textsf{\Large Место для фото}\\[0.5em]\textsf{(#2)}};
			\end{tikzpicture}
		}
		\caption{#2}
		\addcontentsline{lof}{figure}{Фото~\thefigure~—~#2}
	\end{figure}
}

% Приложения (автоматическая нумерация)
\newcounter{appcounter}
\newcommand{\addappendix}[2]{%
	\refstepcounter{appcounter}%
	\begin{table}[H]
		\centering
		\caption*{\textbf{Приложение \theappcounter~— #1}}
		\addcontentsline{lot}{table}{Приложение~\theappcounter~—~#1}
		\begin{tabularx}{0.95\textwidth}{|X|}
			\hline
			#2 \\
			\hline
		\end{tabularx}
	\end{table}
}
